### 3. Working Hypothesis: Failure Modes

**Primary Hypothesis:** The dominant failure mode in a shared-nothing DNET Wasm architecture is **IPC serialization stalls** caused by the Rust host's inability to drain Wasm linear memory buffers faster than sensor ingestion rate, compounded by WASI's copy-in/copy-out semantics on datagram boundaries.

**Observable Symptoms:**
1. Backpressure propagation delays (sensors → Wasm → DNET)
2. Memory linear address space exhaustion under burst loads
3. Gossip protocol thrashing during partition healing

**Predicted Failure Chain:**
```
High-rate sensor burst
    → Wasm linear memory fills (copy-in overhead)
    → IPC channel backpressures
    → Rust host blocks on wasi::io::streams::write
    → UDP dgram drops (no queue space)
    → DNET peers mark node as "slow"
    → Gossip rebalancing amplifies load
    → Cascading degradation
```

**Alternative Hypotheses:**
1. **H2:** WASI capability revocation race conditions during peer churn
2. **H3:** Cross-compilation ABI mismatches between Rust host and Wasm guest on struct layout

//! DNET Edge IPC Router for Wasm-based IoT Processing
//! Target: <50ms P99 latency, shared-nothing peer architecture

use std::collections::HashMap;
use std::sync::atomic::{AtomicU64, Ordering};
use std::time::{Duration, Instant};

use crossbeam_channel::{Sender, Receiver, bounded};
use dashmap::DashMap;
use parking_lot::RwLock;
use smallvec::SmallVec;

/// Maximum latency budget for IPC routing (50ms target)
const LATENCY_BUDGET_MS: u64 = 50;

/// Maximum inline datagram payload before heap allocation
const INLINE_PAYLOAD_SIZE: usize = 128;

/// Peer node identifier (blake3 hash of public key)
#[derive(Clone, Copy, PartialEq, Eq, Hash)]
pub struct PeerId([u8; 32]);

/// Sensor message with zero-copy semantics where possible
#[derive(Debug)]
pub struct SensorMessage {
    /// Source sensor identifier
    pub sensor_id: u64,
    /// Timestamp from sensor (monotonic, not wall clock)
    pub sensor_ts: u64,
    /// Payload - small messages use stack, large use heap
    pub payload: SmallVec<[u8; INLINE_PAYLOAD_SIZE]>,
    /// Routing priority (0 = critical, 255 = best-effort)
    pub priority: u8,
    /// Hop count for TTL-based flooding
    pub hop_count: u8,
}

/// IPC routing table entry
#[derive(Clone)]
pub struct RouteEntry {
    /// Target peer
    pub peer: PeerId,
    /// Estimated RTT in microseconds
    pub rtt_us: AtomicU64,
    /// Last successful transmission
    pub last_seen: RwLock<Instant>,
    /// Whether peer is currently reachable
    pub is_alive: bool,
}

/// Wasm module slot in the processing pool
pub struct WasmSlot {
    /// Module identifier
    pub module_id: u32,
    /// Linear memory base offset (for zero-copy injection)
    pub memory_base: *mut u8,
    /// Current queue depth
    pub queue_depth: AtomicU64,
}

/// The core IPC router - one per edge node
pub struct IpcRouter {
    /// Local peer identity
    local_peer: PeerId,
    
    /// Routing table (content-addressed by PeerId)
    routes: DashMap<PeerId, RouteEntry>,
    
    /// Wasm processing slots
    wasm_pool: RwLock<Vec<WasmSlot>>,
    
    /// Outbound message queue (bounded for backpressure)
    outbound_tx: Sender<RoutedMessage>,
    outbound_rx: Receiver<RoutedMessage>,
    
    /// Inbound message queue from sensors
    inbound_tx: Sender<SensorMessage>,
    inbound_rx: Receiver<SensorMessage>,
    
    /// Metrics for latency tracking
    metrics: RouterMetrics,
    
    /// WASI network capability handle (opaque)
    wasi_network_handle: u32, // wasi:sockets/network::network resource
}

/// Internal routed message with destination
struct RoutedMessage {
    destination: PeerId,
    message: SensorMessage,
    enqueue_time: Instant,
}

/// Performance metrics
#[derive(Default)]
pub struct RouterMetrics {
    pub messages_routed: AtomicU64,
    pub messages_dropped: AtomicU64,
    pub total_latency_us: AtomicU64,
    pub wasm_invocations: AtomicU64,
}

impl IpcRouter {
    /// Create a new IPC router with bounded channels
    pub fn new(local_peer: PeerId, queue_capacity: usize) -> Self {
        let (outbound_tx, outbound_rx) = bounded(queue_capacity);
        let (inbound_tx, inbound_rx) = bounded(queue_capacity);
        
        Self {
            local_peer,
            routes: DashMap::new(),
            wasm_pool: RwLock::new(Vec::with_capacity(16)),
            outbound_tx,
            outbound_rx,
            inbound_tx,
            inbound_rx,
            metrics: RouterMetrics::default(),
            wasi_network_handle: 0,
        }
    }
    
    /// Route a sensor message through the Wasm processing pipeline
    /// Returns Ok(()) if within latency budget, Err if would exceed
    pub fn route(&self, msg: SensorMessage) -> Result<(), RouteError> {
        let start = Instant::now();
        
        // Step 1: Find best Wasm slot (least loaded)
        let slot = self.select_wasm_slot()?;
        
        // Step 2: Check latency budget before processing
        let elapsed_us = start.elapsed().as_micros() as u64;
        let remaining_budget_us = (LATENCY_BUDGET_MS * 1000)
            .saturating_sub(elapsed_us);
        
        if remaining_budget_us < 1000 {
            // Under 1ms remaining - likely to exceed budget
            self.metrics.messages_dropped.fetch_add(1, Ordering::Relaxed);
            return Err(RouteError::LatencyBudgetExceeded);
        }
        
        // Step 3: Copy payload into Wasm linear memory
        // NOTE: This is the critical path - copy-in overhead
        unsafe {
            let dst = slot.memory_base.add(slot.queue_depth.load(Ordering::Relaxed) as usize);
            std::ptr::copy_nonoverlapping(
                msg.payload.as_ptr(),
                dst,
                msg.payload.len().min(INLINE_PAYLOAD_SIZE)
            );
        }
        
        slot.queue_depth.fetch_add(1, Ordering::Relaxed);
        
        // Step 4: Determine next hop via DNET routing
        let next_hop = self.select_next_hop(&msg)?;
        
        // Step 5: Enqueue for transmission
        let routed = RoutedMessage {
            destination: next_hop,
            message: msg,
            enqueue_time: start,
        };
        
        match self.outbound_tx.try_send(routed) {
            Ok(()) => {
                self.metrics.messages_routed.fetch_add(1, Ordering::Relaxed);
                Ok(())
            }
            Err(_) => {
                self.metrics.messages_dropped.fetch_add(1, Ordering::Relaxed);
                Err(RouteError::QueueFull)
            }
        }
    }
    
    /// Select the least-loaded Wasm processing slot
    fn select_wasm_slot(&self) -> Result<WasmSlot, RouteError> {
        let pool = self.wasm_pool.read();
        
        pool.iter()
            .min_by_key(|slot| slot.queue_depth.load(Ordering::Relaxed))
            .cloned()
            .ok_or(RouteError::NoWasmSlots)
    }
    
    /// Select next hop based on DNET gossip state
    fn select_next_hop(&self, msg: &SensorMessage) -> Result<PeerId, RouteError> {
        // Simple greedy routing - select peer with lowest RTT
        // Production would use proper DNET distance metric
        
        self.routes
            .iter()
            .filter(|entry| entry.value().is_alive)
            .min_by_key(|entry| entry.value().rtt_us.load(Ordering::Relaxed))
            .map(|entry| *entry.key())
            .ok_or(RouteError::NoPeers)
    }
    
    /// Register a new peer in the routing table
    pub fn add_peer(&self, peer: PeerId) {
        self.routes.insert(peer, RouteEntry {
            peer,
            rtt_us: AtomicU64::new(10_000), // Initial estimate 10ms
            last_seen: RwLock::new(Instant::now()),
            is_alive: true,
        });
    }
    
    /// Record RTT measurement from gossip
    pub fn update_peer_rtt(&self, peer: &PeerId, rtt_us: u64) {
        if let Some(entry) = self.routes.get(peer) {
            entry.rtt_us.store(rtt_us, Ordering::Relaxed);
            *entry.last_seen.write() = Instant::now();
        }
    }
    
    /// Get current P99 latency estimate
    pub fn p99_latency_us(&self) -> u64 {
        let routed = self.metrics.messages_routed.load(Ordering::Relaxed);
        if routed == 0 {
            return 0;
        }
        
        // Exponential moving average approximation
        let total = self.metrics.total_latency_us.load(Ordering::Relaxed);
        // P99 approximation: multiply mean by 1.3 (typical ratio)
        (total as f64 / routed as f64 * 1.3) as u64
    }
}

#[derive(Debug)]
pub enum RouteError {
    LatencyBudgetExceeded,
    QueueFull,
    NoWasmSlots,
    NoPeers,
    CopyFailed,
}
